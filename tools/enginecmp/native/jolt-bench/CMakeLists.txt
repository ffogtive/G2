# Jolt v5.6.0 네이티브 벤치 (CROSS_PLATFORM_DETERMINISTIC 켬). JOLT_SRC = Jolt 소스 루트
cmake_minimum_required(VERSION 3.20)
project(joltbench CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Distribution)
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release;Distribution")
set(JOLT_SRC "" CACHE PATH "Jolt source root")
option(CPD "CROSS_PLATFORM_DETERMINISTIC" ON)
set(CROSS_PLATFORM_DETERMINISTIC ${CPD} CACHE BOOL "" FORCE)
set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
set(ENABLE_OBJECT_STREAM OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(FLOATING_POINT_EXCEPTIONS_ENABLED OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_FLAGS_DISTRIBUTION "-O3")
add_subdirectory(${JOLT_SRC}/Build jolt)
add_executable(joltbench bench.cpp)
# 벤치 TU 도 Jolt 의 인라인 수학(.inl)을 컴파일하므로 Jolt 와 같은 fp 규칙(FMA 융합 금지)
target_compile_options(joltbench PRIVATE -O3 -ffp-contract=off -fno-rtti -fno-exceptions)
target_link_libraries(joltbench PRIVATE Jolt)
